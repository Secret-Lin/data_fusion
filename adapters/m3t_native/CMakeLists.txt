cmake_minimum_required(VERSION 3.20)
project(mr_nav_m3t LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(M3T_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../thirdparty/3DObjectTracking/M3T")
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
if(MSVC)
  set(OpenMP_CXX_FLAGS "/openmp:experimental" CACHE STRING "M3T uses OpenMP SIMD" FORCE)
endif()
find_package(OpenMP REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc highgui imgcodecs calib3d features2d xfeatures2d)
# Use the vendor's library target without its Windows /MT root settings, hardware
# SDKs, examples or viewers. Conda Python/OpenCV use the shared MSVC runtime.
set(USE_AZURE_KINECT OFF)
set(USE_REALSENSE OFF)
set(USE_GTEST OFF)
add_subdirectory("${M3T_ROOT}/src" vendor)
target_compile_definitions(m3t PUBLIC _USE_MATH_DEFINES NOMINMAX)
if(MSVC)
  target_compile_options(m3t PRIVATE /bigobj)
endif()
pybind11_add_module(_m3t_native bridge.cpp)
target_link_libraries(_m3t_native PRIVATE m3t)
